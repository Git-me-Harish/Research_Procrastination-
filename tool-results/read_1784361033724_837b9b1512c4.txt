    71→@app.post(f"{settings.API_V1_PREFIX}/auth/register", response_model=TokenOut, status_code=201)
    72→def register(payload: UserCreate, db: Session = Depends(get_db)):
    73→    if db.query(User).filter((User.email == payload.email) | (User.username == payload.username)).first():
    74→        raise HTTPException(status_code=400, detail="Email or username already registered")
    75→    user = User(
    76→        email=payload.email,
    77→        username=payload.username,
    78→        hashed_password=hash_password(payload.password),
    79→        display_name=payload.display_name or payload.username,
    80→    )
    81→    db.add(user)
    82→    db.commit()
    83→    db.refresh(user)
    84→    token = create_access_token(str(user.id))
    85→    return TokenOut(access_token=token, user=UserOut.model_validate(user))

    88→@app.post(f"{settings.API_V1_PREFIX}/auth/login", response_model=TokenOut)
    89→def login(payload: UserLogin, db: Session = Depends(get_db)):
    90→    user = db.query(User).filter(User.email == payload.email).first()
    91→    if not user or not verify_password(payload.password, user.hashed_password):
    92→        raise HTTPException(status_code=401, detail="Invalid credentials")
    93→    if not user.is_active:
    94→        raise HTTPException(status_code=403, detail="Account deactivated")
    95→    token = create_access_token(str(user.id))
    96→    return TokenOut(access_token=token, user=UserOut.model_validate(user))

   138→@app.get(f"{settings.API_V1_PREFIX}/profile/summary", response_model=ProfileSummary)

   141→    db: Session = Depends(get_db),
   142→):
   143→    """Aggregated profile data — lifetime totals, level progress, recent activity."""
   144→    # ---- Level progress ----
   145→    cur_level_xp = xp_for_level(user.level)
   146→    next_level = user.level + 1
   147→    next_level_xp = xp_for_level(next_level)
   148→    xp_into_level = user.xp - cur_level_xp
   149→    xp_for_next_level = next_level_xp - cur_level_xp
   150→    xp_to_next_level = max(0, next_level_xp - user.xp)
   151→    progress_pct = (xp_into_level / xp_for_next_level * 100) if xp_for_next_level > 0 else 100
   152→    progress_pct = max(0, min(100, progress_pct))
   153→
   154→    # ---- Task totals ----
   155→    total_tasks = db.query(func.count(Task.id)).filter(Task.user_id == user.id).scalar() or 0
   156→    tasks_completed = db.query(func.count(Task.id)).filter(
   157→        Task.user_id == user.id, Task.status == TaskStatus.COMPLETED
   158→    ).scalar() or 0
   159→    tasks_pending = total_tasks - tasks_completed
   160→
   161→    # ---- Focus totals ----
   162→    focus_agg = db.query(
   163→        func.coalesce(func.sum(FocusSession.actual_minutes), 0),
   164→        func.count(FocusSession.id),
   165→    ).filter(
   166→        FocusSession.user_id == user.id,
   167→        FocusSession.status == SessionStatus.COMPLETED
   168→    ).one()
   169→    total_focus_minutes = int(focus_agg[0] or 0)
   170→    total_focus_sessions = int(focus_agg[1] or 0)
   171→
   172→    # ---- Breathe totals ----
   173→    breathe_agg = db.query(
   174→        func.count(BreatheSession.id),
   175→        func.coalesce(func.sum(BreatheSession.duration_seconds), 0),
   176→    ).filter(BreatheSession.user_id == user.id).one()
   177→    total_breathe_sessions = int(breathe_agg[0] or 0)
   178→    total_breathe_minutes = int((breathe_agg[1] or 0) // 60)
   179→
   180→    # ---- Chain totals ----
   181→    chain_agg = db.query(
   182→        func.count(PowerChain.id),
   183→        func.coalesce(func.sum(PowerChain.total_completions), 0),
   184→        func.coalesce(func.max(PowerChain.longest_chain_days), 0),
   185→    ).filter(PowerChain.user_id == user.id).one()
   186→    total_chains = int(chain_agg[0] or 0)
   187→    total_chain_completions = int(chain_agg[1] or 0)
   188→    longest_chain = int(chain_agg[2] or 0)
   189→
   190→    # ---- Shield totals ----
   191→    shield_agg = db.query(
   192→        func.count(StreakShield.id),
   193→        func.coalesce(func.sum(StreakShield.is_spent.cast(Integer)), 0),
   194→    ).filter(StreakShield.user_id == user.id).one()
   195→    total_shields_earned = int(shield_agg[0] or 0)
   196→    total_shields_spent = int(shield_agg[1] or 0)
   197→
   198→    # ---- Achievements ----
   199→    achievements_earned = db.query(func.count(UserAchievement.id)).filter(
   200→        UserAchievement.user_id == user.id
   201→    ).scalar() or 0
   202→    achievements_total = db.query(func.count(Achievement.id)).scalar() or 0
   203→
   204→    # ---- Mood + AI ----
   205→    mood_entries = db.query(func.count(MoodEntry.id)).filter(
   206→        MoodEntry.user_id == user.id
   207→    ).scalar() or 0
   208→    ai_interactions = db.query(func.count(AIInteraction.id)).filter(
   209→        AIInteraction.user_id == user.id
   210→    ).scalar() or 0
   211→
   212→    # ---- Days active (distinct dates from all activity tables) ----
   213→    task_dates = db.query(func.distinct(func.date(Task.created_at))).filter(
   214→        Task.user_id == user.id
   215→    ).all()
   216→    focus_dates = db.query(func.distinct(func.date(FocusSession.started_at))).filter(
   217→        FocusSession.user_id == user.id
   218→    ).all()
   219→    mood_dates = db.query(func.distinct(func.date(MoodEntry.created_at))).filter(
   220→        MoodEntry.user_id == user.id
   221→    ).all()
   222→    breathe_dates = db.query(func.distinct(func.date(BreatheSession.created_at))).filter(
   223→        BreatheSession.user_id == user.id
   224→    ).all()
   225→    all_dates = set()
   226→    for rows in (task_dates, focus_dates, mood_dates, breathe_dates):
   227→        for (d,) in rows:
   228→            if d:
   229→                all_dates.add(str(d))
   230→    days_active = len(all_dates)
   231→
   232→    # ---- Recent activity feed (merge last 15 across sources) ----
   233→    activity: list[ProfileActivityItem] = []
   234→
   235→    # Tasks completed
   236→    for t in db.query(Task).filter(
   237→        Task.user_id == user.id, Task.status == TaskStatus.COMPLETED, Task.completed_at.isnot(None)
   238→    ).order_by(Task.completed_at.desc()).limit(10).all():
   239→        activity.append(ProfileActivityItem(
   240→            kind="task_completed",
   241→            title=f"Completed: {t.title}",
   242→            detail=f"{t.actual_minutes or 0}m actual / {t.estimated_minutes or 0}m est",
   243→            timestamp=t.completed_at,
   244→            xp=25 + (t.difficulty or 2) * 10 + (t.estimated_minutes or 25),
   245→        ))
   246→
   247→    # Focus sessions completed
   248→    for s in db.query(FocusSession).filter(
   249→        FocusSession.user_id == user.id, FocusSession.status == SessionStatus.COMPLETED
   250→    ).order_by(FocusSession.ended_at.desc()).limit(10).all():
   251→        activity.append(ProfileActivityItem(
   252→            kind="focus_session",
   253→            title=f"{s.session_type.replace('_',' ').title()} — {s.actual_minutes}m",
   254→            detail=f"Quality {s.focus_quality}/5 · Distractions {s.distractions}",
   255→            timestamp=s.ended_at or s.started_at,
   256→            xp=(s.actual_minutes or 0) * (3 if s.session_type == SessionType.DEEP_WORK else 2),
   257→        ))
   258→
   259→    # Mood logs
   260→    for m in db.query(MoodEntry).filter(MoodEntry.user_id == user.id).order_by(
   261→        MoodEntry.created_at.desc()
   262→    ).limit(10).all():
   263→        activity.append(ProfileActivityItem(
   264→            kind="mood_logged",
   265→            title=f"Mood {m.mood_score}/5 · Energy {m.energy_score}/5",
   266→            detail=", ".join(m.triggers) if m.triggers else None,
   267→            timestamp=m.created_at,
   268→            xp=5,
   269→        ))
   270→
   271→    # Breathe sessions
   272→    for b in db.query(BreatheSession).filter(BreatheSession.user_id == user.id).order_by(
   273→        BreatheSession.created_at.desc()
   274→    ).limit(10).all():
   275→        activity.append(ProfileActivityItem(
   276→            kind="breathe_session",
   277→            title=f"Breathe: {b.technique.replace('_','-')} · {b.cycles_completed} cycles",
   278→            detail=f"Calm {b.calmness_before}→{b.calmness_after} · {b.duration_seconds}s",
   279→            timestamp=b.created_at,
   280→            xp=b.xp_earned or 0,
   281→        ))
   282→
   283→    # Chain completions (use last_completed_date on chains)
   284→    for c in db.query(PowerChain).filter(
   285→        PowerChain.user_id == user.id, PowerChain.last_completed_date.isnot(None)
   286→    ).order_by(PowerChain.last_completed_date.desc()).limit(10).all():
   287→        ts = datetime.strptime(c.last_completed_date, "%Y-%m-%d").replace(tzinfo=timezone.utc)
   288→        activity.append(ProfileActivityItem(
   289→            kind="chain_completed",
   290→            title=f"Chain day: {c.title}",
   291→            detail=f"Day {c.current_chain_days} (best: {c.longest_chain_days})",
   292→            timestamp=ts,
   293→            xp=50,
   294→        ))
   295→
   296→    # Shield earned
   297→    for sh in db.query(StreakShield).filter(StreakShield.user_id == user.id).order_by(
   298→        StreakShield.earned_at.desc()
   299→    ).limit(10).all():
   300→        activity.append(ProfileActivityItem(
   301→            kind="shield_earned",
   302→            title=f"{sh.rarity.title()} {sh.shield_color.title()} shield",
   303→            detail=sh.source_detail or sh.source,
   304→            timestamp=sh.earned_at,
   305→            xp=0,
   306→        ))
   307→
   308→    # Achievement earned
   309→    for ua in db.query(UserAchievement).filter(UserAchievement.user_id == user.id).order_by(
   310→        UserAchievement.earned_at.desc()
   311→    ).limit(10).all():
   312→        activity.append(ProfileActivityItem(
   313→            kind="achievement_earned",
   314→            title=f"Trophy: {ua.achievement.title}",
   315→            detail=ua.achievement.description,
   316→            timestamp=ua.earned_at,
   317→            xp=ua.achievement.xp_reward or 0,
   318→        ))
   319→
   320→    # AI interactions
   321→    for ai in db.query(AIInteraction).filter(AIInteraction.user_id == user.id).order_by(
   322→        AIInteraction.created_at.desc()
   323→    ).limit(10).all():
   324→        activity.append(ProfileActivityItem(
   325→            kind="ai_interaction",
   326→            title=f"AI {ai.interaction_type}",
   327→            detail=ai.model_used,
   328→            timestamp=ai.created_at,

   331→
   332→    # Sort by timestamp desc and trim to 15
   333→    activity.sort(key=lambda a: a.timestamp, reverse=True)
   334→    activity = activity[:15]
   335→
   336→    return ProfileSummary(
   337→        user=user,
   338→        level=user.level,
   339→        xp=user.xp,
   340→        xp_into_level=xp_into_level,
   341→        xp_for_next_level=xp_for_next_level,
   342→        xp_to_next_level=xp_to_next_level,
   343→        next_level=next_level,
   344→        progress_pct=progress_pct,
   345→        total_tasks=total_tasks,
   346→        tasks_completed=tasks_completed,
   347→        tasks_pending=tasks_pending,
   348→        total_focus_minutes=total_focus_minutes,
   349→        total_focus_sessions=total_focus_sessions,
   350→        total_breathe_sessions=total_breathe_sessions,
   351→        total_breathe_minutes=total_breathe_minutes,
   352→        total_chains=total_chains,
   353→        total_chain_completions=total_chain_completions,
   354→        longest_chain=longest_chain,
   355→        total_shields_earned=total_shields_earned,
   356→        total_shields_spent=total_shields_spent,
   357→        achievements_earned=achievements_earned,
   358→        achievements_total=achievements_total,
   359→        mood_entries=mood_entries,

   458→@app.patch(f"{settings.API_V1_PREFIX}/tasks/{{task_id}}", response_model=TaskOut)
   459→def update_task(
   460→    task_id: int,
   461→    payload: TaskUpdate,
   462→    user: User = Depends(get_current_user),
   463→    db: Session = Depends(get_db),
   464→):
   465→    task = db.query(Task).filter(Task.id == task_id, Task.user_id == user.id).first()
   466→    if not task:
   467→        raise HTTPException(status_code=404, detail="Task not found")
   468→
   469→    was_completed = task.status == TaskStatus.COMPLETED
   470→    update_data = payload.model_dump(exclude_unset=True)
   471→
   472→    if "status" in update_data:
   473→        new_status = TaskStatus(update_data["status"])
   474→        task.status = new_status
   475→        if new_status == TaskStatus.COMPLETED and not was_completed:
   476→            task.completed_at = datetime.now(timezone.utc)
   477→            # Award XP based on difficulty
   478→            xp = 25 + (task.difficulty or 2) * 10 + (task.estimated_minutes or 25)
   479→            xp_summary = add_xp(user, xp, db)
   480→            update_streak(user, db)
   481→            new_ach = evaluate_achievements(user, db)
   482→            update_data["_xp_awarded"] = xp
   483→            update_data["_new_achievements"] = new_ach
   484→        elif was_completed and new_status != TaskStatus.COMPLETED:
   485→            task.completed_at = None
   486→
   487→    for k in ("title", "description", "priority", "category",
   488→              "estimated_minutes", "actual_minutes", "due_date",
   489→              "breakdown", "energy_required", "difficulty"):
   490→        if k in update_data:
   491→            setattr(task, k, update_data[k])
   492→
   493→    db.commit()
   494→    db.refresh(task)
   495→    return task

   900→@app.post(f"{settings.API_V1_PREFIX}/chains/{{chain_id}}/links/{{link_id}}/toggle",
   901→          response_model=ChainLinkToggleOut)
   902→def toggle_link(
   903→    chain_id: int,
   904→    link_id: int,
   905→    user: User = Depends(get_current_user),
   906→    db: Session = Depends(get_db),
   907→):
   908→    """Toggle a chain link's completion for today.
   909→
   910→    If all required links become completed, the chain day is incremented and the user
   911→    earns XP + a chance for a shield. If a previously-completed link is un-toggled
   912→    after the day was marked complete, the chain breaks (with SNAP!).
   913→    """
   914→    chain = db.query(PowerChain).filter(
   915→        PowerChain.id == chain_id, PowerChain.user_id == user.id
   916→    ).first()
   917→    if not chain:
   918→        raise HTTPException(404, "Chain not found")
   919→
   920→    link = db.query(ChainLink).filter(
   921→        ChainLink.id == link_id, ChainLink.chain_id == chain_id
   922→    ).first()
   923→    if not link:
   924→        raise HTTPException(404, "Link not found")
   925→
   926→    today = datetime.now(timezone.utc).date().isoformat()
   927→    was_chain_complete_today = chain.last_completed_date == today
   928→    new_state = not link.completed_today
   929→    link.completed_today = new_state
   930→
   931→    xp_earned = 0
   932→    new_shield = None
   933→    chain_broken = False
   934→
   935→    if new_state:
   936→        # Marking complete
   937→        link.last_completed_date = today
   938→        link.total_completions = (link.total_completions or 0) + 1
   939→        xp_earned = 15
   940→        # Check if all required links are now complete
   941→        required_links = [l for l in chain.links if l.is_required]
   942→        all_complete = all(l.completed_today for l in required_links)
   943→        if all_complete and not was_chain_complete_today:
   944→            # Chain day complete!
   945→            chain.current_chain_days = (chain.current_chain_days or 0) + 1
   946→            chain.total_completions = (chain.total_completions or 0) + 1
   947→            chain.last_completed_date = today
   948→            if chain.current_chain_days > (chain.longest_chain_days or 0):
   949→                chain.longest_chain_days = chain.current_chain_days
   950→            xp_earned += 50
   951→            # Bonus: every 3-day chain = award a shield
   952→            if chain.current_chain_days % 3 == 0:
   953→                shield_color = "gold" if chain.current_chain_days >= 9 else "blue"
   954→                rarity = "rare" if chain.current_chain_days >= 9 else "common"
   955→                new_shield = award_shield(
   956→                    user, "power_chain",
   957→                    f"Chain '{chain.title}' reached {chain.current_chain_days} days",
   958→                    db, color=shield_color, rarity=rarity
   959→                )
   960→                new_shield = {
   961→                    "id": new_shield.id,
   962→                    "shield_color": new_shield.shield_color,
   963→                    "rarity": new_shield.rarity,
   964→                    "source": "power_chain",
   965→                }
   966→    else:
   967→        # Un-toggling — if chain was complete today, break it
   968→        link.last_completed_date = None
   969→        if was_chain_complete_today:
   970→            chain_broken = True
   971→            chain.current_chain_days = 0
   972→            chain.broken_at = datetime.now(timezone.utc)
   973→            chain.is_active = False
   974→
   975→    add_xp(user, xp_earned, db)
   976→    db.commit()
   977→    db.refresh(link)
   978→    db.refresh(chain)
   979→
   980→    return ChainLinkToggleOut(
   981→        link=link,
   982→        chain_completed_today=chain.last_completed_date == today,
   983→        chain_broken=chain_broken,
   984→        xp_earned=xp_earned,
   985→        new_shield_earned=new_shield,

  1038→@app.post(f"{settings.API_V1_PREFIX}/shields/spend", response_model=ShieldSpendResponse)
  1039→def spend_shield(
  1040→    payload: ShieldSpendRequest,
  1041→    user: User = Depends(get_current_user),
  1042→    db: Session = Depends(get_db),
  1043→):
  1044→    """Spend a shield to protect a missed day (target_date must be yesterday or older)."""
  1045→    try:
  1046→        target = datetime.strptime(payload.target_date, "%Y-%m-%d").date()
  1047→    except ValueError:
  1048→        raise HTTPException(400, "target_date must be YYYY-MM-DD")
  1049→
  1050→    today = datetime.now(timezone.utc).date()
  1051→    if target >= today:
  1052→        raise HTTPException(400, "Can only protect past days (yesterday or earlier).")
  1053→
  1054→    # Check if user already has a shield spent for this date
  1055→    existing = db.query(StreakShield).filter(
  1056→        StreakShield.user_id == user.id,
  1057→        StreakShield.spent_for_date == payload.target_date
  1058→    ).first()
  1059→    if existing:
  1060→        return ShieldSpendResponse(
  1061→            success=False, message="This date is already protected.",
  1062→            streak_protected=True
  1063→        )
  1064→
  1065→    # Find an unspent shield
  1066→    shield = db.query(StreakShield).filter(
  1067→        StreakShield.user_id == user.id,
  1068→        StreakShield.is_spent == False
  1069→    ).order_by(StreakShield.earned_at.asc()).first()
  1070→    if not shield:
  1071→        raise HTTPException(400, "No shields in reserve! Earn more by completing power chains or deep work sessions.")
  1072→
  1073→    shield.is_spent = True
  1074→    shield.spent_at = datetime.now(timezone.utc)
  1075→    shield.spent_for_date = payload.target_date
  1076→
  1077→    # Update streak — pretend last_active_date is today so streak doesn't break
  1078→    user.last_active_date = today.isoformat()
  1079→    if user.current_streak == 0:
  1080→        user.current_streak = 1
  1081→
  1082→    db.commit()
  1083→    db.refresh(shield)
  1084→    return ShieldSpendResponse(
  1085→        success=True, shield=shield,
  1086→        message=f"Shield spent! {payload.target_date} is now protected. Your streak is safe.",
  1087→        streak_protected=True
  1088→    )
